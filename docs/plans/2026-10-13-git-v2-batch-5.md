```rust
//! History's graph: each commit's lane and the lines into and out of its row, from the commits
//! (children before parents: `--date-order`) and their parents. Pure: tested on lists.
//! See docs/specs/git.md, section 6.

use crate::parse::Commit;

/// One row of the graph. A line is (its lane at one edge, its lane at the other): `up` from the
/// row's top to its middle, `down` from its middle to its bottom.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct GraphRow {
    /// The commit's lane: where its dot is.
    pub lane: usize,
    pub up: Vec<(usize, usize)>,
    pub down: Vec<(usize, usize)>,
    /// The lanes the row draws in (its widest one, plus one).
    pub width: usize,
}

/// The first free lane, or a new one.
fn free(lanes: &mut Vec<Option<&str>>) -> usize {
    match lanes.iter().position(Option::is_none) {
        Some(i) => i,
        None => {
            lanes.push(None);
            lanes.len() - 1
        }
    }
}

/// The rows for `commits`, newest first. Each lane waits for the commit it leads to. A commit
/// takes the first lane waiting for it (a branch's tip, which none waits for, the first free
/// one); the other lanes waiting for it join it there. Its first parent goes on in its lane,
/// each other parent into a lane already waiting for it, else a free one. Every call starts from
/// nothing, so Load more (the whole list again) leaves the earlier rows as they were.
pub fn graph(commits: &[Commit]) -> Vec<GraphRow> {
    let mut lanes: Vec<Option<&str>> = Vec::new();
    let mut rows = Vec::with_capacity(commits.len());
    for c in commits {
        let me = c.hash.as_str();
        let lane = match lanes.iter().position(|l| *l == Some(me)) {
            Some(i) => i,
            None => free(&mut lanes),
        };
        let mut row = GraphRow {
            lane,
            ..GraphRow::default()
        };
        // The lanes passing by this commit, waiting for another one.
        let mut passing = Vec::new();
        for (i, l) in lanes.iter().enumerate() {
            match *l {
                Some(h) if h == me => row.up.push((i, lane)),
                Some(_) => {
                    row.up.push((i, i));
                    passing.push(i);
                }
                None => {}
            }
        }
        for l in lanes.iter_mut() {
            if *l == Some(me) {
                *l = None;
            }
        }
        let mut parents = c.parents.iter().map(String::as_str);
        lanes[lane] = parents.next();
        if lanes[lane].is_some() {
            row.down.push((lane, lane));
        }
        for p in parents {
            let to = match lanes.iter().position(|l| *l == Some(p)) {
                Some(i) => i,
                None => {
                    let i = free(&mut lanes);
                    lanes[i] = Some(p);
                    i
                }
            };
            row.down.push((lane, to));
        }
        row.down.extend(passing.iter().map(|&i| (i, i)));
        while lanes.last() == Some(&None) {
            lanes.pop();
        }
        row.width = row
            .up
            .iter()
            .chain(&row.down)
            .flat_map(|&(a, b)| [a, b])
            .chain([lane])
            .max()
            .unwrap_or(0)
            + 1;
        rows.push(row);
    }
    rows
}
```

```rust
/// A commit with no change, its dates set (seconds since 1970).
fn commit_at(repo: &Repo, message: &str, seconds: u64) {
    let date = format!("{seconds} +0000");
    let ok = std::process::Command::new("git")
        .args(["commit", "-q", "--allow-empty", "-m", message])
        .current_dir(&repo.root)
        .env("GIT_COMMITTER_DATE", &date)
        .env("GIT_AUTHOR_DATE", &date)
        .status()
        .unwrap()
        .success();
    assert!(ok);
}
```

```rust
    records(out).filter_map(commit_of).collect()

/// `git log -z --follow --name-status --format=LOG_FORMAT -- path`: each commit with the file
/// as it was in it (a rename's old path in `from`). After each record git prints "\n" and the
/// status, then one path (two for a rename or a copy); a merge prints none, and keeps the path
/// of the commit after it in the list.
pub fn parse_file_log(out: &[u8], path: &str) -> Vec<(Commit, CommitFile)> {
    let mut entries = Vec::new();
    let mut now = path.to_string();
    let mut fields = records(out).peekable();
    while let Some(record) = fields.next() {
        let Some(commit) = commit_of(record.trim_start_matches('\n')) else {
            continue;
        };
        let mut file = CommitFile {
            change: Change::Modified,
            path: now.clone(),
            from: None,
        };
        let status = fields
            .peek()
            .filter(|f| !f.contains('\x1f'))
            .map(|f| f.trim_start_matches('\n').to_string());
        if let Some(code) = status {
            fields.next();
            if let Some(change) = code.bytes().next().and_then(Change::of) {
                let first = fields.next().unwrap_or("").to_string();
                file = if change == Change::Renamed || code.starts_with('C') {
                    CommitFile {
                        change,
                        path: fields.next().unwrap_or("").to_string(),
                        from: Some(first),
                    }
                } else {
                    CommitFile {
                        change,
                        path: first,
                        from: None,
                    }
                };
            }
        }
        // Older commits knew it by its name before this one's rename.
        now = file.from.clone().unwrap_or_else(|| file.path.clone());
        entries.push((commit, file));
    }
    entries
}
```

```rust
pub enum Search {
    /// Commit messages (`--grep`), without case.
    #[default]
    Message,
    /// Authors' names and emails (`--author`), without case.
    Author,
    /// Commits that added or removed the text (`-S`).
    Code,
}

/// What History lists.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct HistoryQuery {
    /// HEAD and every branch, tag and remote branch (not `--all`, which has the stashes); else
    /// HEAD's history.
    pub all: bool,
    pub search: Option<(Search, String)>,
}

impl Repo {
    /// `count` commits of the query after the first `skip`, in `--date-order` (children before
    /// parents, so the graph's lanes never cross back; every page shares it). Searches take the
    /// text as written (`-F`: a bracket is a bracket). Before the first commit: none.
    pub fn history(&self, q: &HistoryQuery, skip: usize, count: usize) -> Result<Vec<Commit>, GitError> {
        let format = format!("--format={}", parse::LOG_FORMAT);
        let date = format!("--date={}", parse::LOG_DATE);
        let (skip, count) = (format!("--skip={skip}"), format!("-n{count}"));
        let search: Vec<String> = match &q.search {
            Some((Search::Message, t)) => vec!["-i".into(), "-F".into(), format!("--grep={t}")],
            Some((Search::Author, t)) => vec!["-i".into(), "-F".into(), format!("--author={t}")],
            Some((Search::Code, t)) => vec![format!("-S{t}")],
            None => Vec::new(),
        };
        let mut args = vec!["log", "-z", &format, &date, "--date-order", &skip, &count];
        if q.all {
            args.extend(["HEAD", "--branches", "--tags", "--remotes"]);
        }
        args.extend(search.iter().map(String::as_str));
        args.push("--");
        match self.run(&args) {
            Ok(out) => Ok(parse::parse_log(&out)),
            Err(e) if e.stderr.contains("does not have any commits") || e.stderr.contains("unknown revision") => {
                Ok(Vec::new())
            }
            Err(e) => Err(e),
        }
    }

    /// One commit, as History lists it (Blame opens one that isn't loaded).
    pub fn commit(&self, hash: &str) -> Result<Commit, GitError> {
        let format = format!("--format={}", parse::LOG_FORMAT);
        let date = format!("--date={}", parse::LOG_DATE);
        let out = self.run(&["log", "-z", "-1", &format, &date, hash, "--"])?;
        parse::parse_log(&out).pop().ok_or_else(|| GitError {
            command: shown(&self.program, &["log", "-1", hash]),
            code: None,
            stderr: format!("fatal: no commit {hash}"),
            missing: false,
        })
    }

    /// HEAD's commits not on its upstream (`git rev-list @{u}..HEAD`): History's "not pushed".
    /// None with no upstream, or one gone from the server.
    pub fn unpushed(&self) -> Result<HashSet<String>, GitError> {
        if self.run(&["rev-parse", "--verify", "-q", "@{u}"]).is_err() {
            return Ok(HashSet::new());
        }
        Ok(self.text(&["rev-list", "@{u}..HEAD"])?.lines().map(str::to_string).collect())
    }

    /// `path`'s history through its renames (`--follow`), each commit with the file's path then.
    pub fn file_log(&self, path: &str, skip: usize, count: usize) -> Result<Vec<(Commit, CommitFile)>, GitError> {
        let format = format!("--format={}", parse::LOG_FORMAT);
        let date = format!("--date={}", parse::LOG_DATE);
        let (skip, count) = (format!("--skip={skip}"), format!("-n{count}"));
        let out = self.run(&[
            "-c", "core.quotePath=false", "log", "-z", &format, &date, "--follow", "--name-status", "-M", &skip, &count,
            "--", path,
        ])?;
        Ok(parse::parse_file_log(&out, path))
    }

    /// `path` as it was in `rev`, as a checkout would write it: `git cat-file --filters` applies
    /// the working tree's line endings and filters to the stored bytes.
    pub fn file_at(&self, rev: &str, path: &str) -> Result<Vec<u8>, GitError> {
        self.run(&["cat-file", "--filters", &format!("{rev}:{path}")])
    }
}
```

```rust
//! `git blame --porcelain`, parsed, and as the diff panel shows it: one group of rows per run
//! of lines from one commit, under a heading naming it. See docs/specs/git.md, section 6, Blame.

use std::collections::HashMap;

use crate::diff::{FileDiff, Hunk, Kind, Line};
use crate::parse::ago;

/// One line of the file and the commit that last changed it.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct BlameLine {
    /// Forty (or sixty-four) zeros for a line not committed yet.
    pub hash: String,
    pub author: String,
    /// Unix seconds.
    pub time: i64,
    pub summary: String,
    /// As the file has it, without its '\n' (a CRLF line keeps its '\r').
    pub text: String,
}

/// The porcelain format: a header `<hash> <orig> <final> [<count>]`, the commit's details the
/// first time it shows (`author`, `author-time`, `summary`, …), then the line after a TAB.
pub fn parse_blame(out: &[u8]) -> Vec<BlameLine> {
    let text = String::from_utf8_lossy(out);
    let mut known: HashMap<String, (String, i64, String)> = HashMap::new();
    let mut hash = String::new();
    let mut lines = Vec::new();
    for l in text.split('\n') {
        if let Some(content) = l.strip_prefix('\t') {
            let (author, time, summary) = known.get(&hash).cloned().unwrap_or_default();
            lines.push(BlameLine {
                hash: hash.clone(),
                author,
                time,
                summary,
                text: content.to_string(),
            });
            continue;
        }
        let Some((key, value)) = l.split_once(' ') else {
            continue;
        };
        match key {
            "author" => known.entry(hash.clone()).or_default().0 = value.to_string(),
            "author-time" => known.entry(hash.clone()).or_default().1 = value.parse().unwrap_or(0),
            "summary" => known.entry(hash.clone()).or_default().2 = value.to_string(),
            k if k.len() >= 40 && k.bytes().all(|b| b.is_ascii_hexdigit()) => hash = k.to_string(),
            _ => {}
        }
    }
    lines
}

/// The panel's view of a blame: a hunk per run of lines from one commit, its heading
/// "a1b2c3d · Ada Lovelace · 3 d ago · Subject", its lines numbered as in the file; and each
/// hunk's commit, for a click on its heading.
pub fn blame_diff(path: &str, lines: &[BlameLine], now: i64) -> (FileDiff, Vec<String>) {
    let mut file = FileDiff::empty();
    file.old_path = Some(path.to_string());
    file.new_path = Some(path.to_string());
    let mut hashes: Vec<String> = Vec::new();
    for (i, l) in lines.iter().enumerate() {
        let n = i as u32 + 1;
        if hashes.last() != Some(&l.hash) {
            let heading = if l.hash.bytes().all(|b| b == b'0') {
                "Not committed yet".to_string()
            } else {
                format!("{} · {} · {} · {}", &l.hash[..7], l.author, ago(now, l.time), l.summary)
            };
            file.hunks.push(Hunk {
                old_start: 0,
                old_len: 0,
                new_start: n,
                new_len: 0,
                context: heading,
                lines: Vec::new(),
            });
            hashes.push(l.hash.clone());
        }
        let h = file.hunks.last_mut().expect("a hunk was just made");
        h.lines.push(Line {
            kind: Kind::Context,
            old_no: None,
            new_no: Some(n),
            text: l.text.clone(),
            no_newline: false,
        });
        h.new_len += 1;
    }
    (file, hashes)
}
```

```rust
    /// A commit clicked (in the list, or a blame's group): the left column shows its message
    /// and files, the panel its whole diff (against its first parent).
    pub(crate) fn show_commit(&mut self, id: BlockId, hash: String, window: &mut Window, cx: &mut Context<Self>) {
        let Some(git) = self.space.git(id) else {
            return;
        };
        let Some(repo) = git.repo().cloned() else {
            return;
        };
        let loaded = git.commits.iter().find(|c| c.hash == hash).cloned();
        let ignore_ws = git.history_panel.ignore_ws;
        self.git_job(
            self.here(id),
            move || -> Result<(Commit, Vec<CommitFile>, Vec<FileDiff>), GitError> {
                let commit = match loaded {
                    Some(c) => c,
                    None => repo.commit(&hash)?,
                };
                Ok((commit, repo.commit_files(&hash)?, repo.commit_changes(&hash, ignore_ws)?))
            },
            move |ws, at, read, _, cx| {
                let Some(git) = ws.git_at(&at) else {
                    return;
                };
                match read {
                    Ok((commit, files, diffs)) => {
                        git.history_panel.source = Source::Range {
                            title: format!("Commit {}", commit.short),
                        };
                        git.history_panel.set_files(diffs);
                        git.shown = Some(git_block::Shown { commit, files, file: None });
                        git.error = None;
                    }
                    Err(e) => git.error = Some(e.to_string()),
                }
                cx.notify();
            },
            window,
            cx,
        );
    }

    /// The commit column's ←: back to the list, the panel empty.
    pub(crate) fn hide_commit(&mut self, id: BlockId, cx: &mut Context<Self>) {
        if let Some(git) = self.space.git_mut(id) {
            git.shown = None;
            git.history_panel.set_files(Vec::new());
            cx.notify();
        }
    }

    /// A file of the commit shown: the panel scrolls to it.
    pub(crate) fn show_commit_file(&mut self, id: BlockId, path: String, cx: &mut Context<Self>) {
        let Some(git) = self.space.git_mut(id) else {
            return;
        };
        if let Some(ix) = git.history_panel.row_of(false, &path) {
            git.history_panel.scroll.scroll_to_item(ix, gpui_kit::ScrollStrategy::Top);
        }
        if let Some(shown) = git.shown.as_mut() {
            shown.file = Some(path);
        }
        cx.notify();
    }
```

```rust
/// What History lists now (Task 7 adds the search).
fn history_query(git: &Git) -> HistoryQuery {
    HistoryQuery {
        all: git.all_branches,
        search: None,
    }
}

    /// Loads a page of History: the first replaces the list (and reads which commits aren't
    /// pushed), a later one extends it; the graph is laid out again over the whole list, so its
    /// lanes go on. A result for a query changed meanwhile is dropped.
    pub(crate) fn load_history(&mut self, at: GitAt, skip: usize, window: &mut Window, cx: &mut Context<Self>) {
        let Some(git) = self.git_at(&at) else {
            return;
        };
        let Some(repo) = git.repo().cloned() else {
            return;
        };
        let q = history_query(git);
        let asked = q.clone();
        self.git_job(
            at,
            move || -> Result<(Vec<Commit>, Option<HashSet<String>>), GitError> {
                let page = repo.history(&q, skip, git_block::PAGE)?;
                let unpushed = if skip == 0 { Some(repo.unpushed()?) } else { None };
                Ok((page, unpushed))
            },
            move |ws, at, read, _, _| {
                let Some(git) = ws.git_at(&at) else {
                    return;
                };
                if history_query(git) != asked {
                    return;
                }
                match read {
                    Ok((page, unpushed)) => {
                        git.more = page.len() == git_block::PAGE;
                        let mut all = if skip == 0 { Vec::new() } else { (*git.commits).clone() };
                        all.extend(page);
                        git.graph = Rc::new(if asked.search.is_none() { graph(&all) } else { Vec::new() });
                        git.commits = Rc::new(all);
                        if let Some(u) = unpushed {
                            git.unpushed = Rc::new(u);
                        }
                        git.error = None;
                    }
                    Err(e) => git.error = Some(e.to_string()),
                }
            },
            window,
            cx,
        );
    }

    /// All branches on or off: the list read again.
    pub(crate) fn toggle_all_branches(&mut self, id: BlockId, window: &mut Window, cx: &mut Context<Self>) {
        let Some(git) = self.space.git_mut(id) else {
            return;
        };
        git.all_branches = !git.all_branches;
        git.shown = None;
        cx.notify();
        let at = self.here(id);
        self.load_history(at, 0, window, cx);
    }
```

```rust
/// A row's graph: its lines (top to middle, middle to bottom) and its commit's dot, each lane
/// in a color of the terminal's palette, as wide as the widest row's lanes (at most MAX_LANES).
fn graph_cell(row: &GraphRow, lanes: usize, palette: [Hsla; 6], z: f32) -> impl IntoElement {
    let row = row.clone();
    canvas(
        |_, _, _| {},
        move |bounds, _, window, _| {
            let lane_w = LANE_W * z;
            let x = |lane: usize| bounds.origin.x + px(lane_w * (lane as f32 + 0.5));
            let top = bounds.origin.y;
            let mid = top + bounds.size.height / 2.0;
            let bottom = top + bounds.size.height;
            let color = |lane: usize| palette[lane % palette.len()];
            let lines = row
                .up
                .iter()
                .map(|&(a, b)| (a, b, top, mid, a))
                .chain(row.down.iter().map(|&(a, b)| (a, b, mid, bottom, b)));
            for (a, b, y0, y1, hue) in lines {
                if a >= MAX_LANES || b >= MAX_LANES {
                    continue;
                }
                let mut path = PathBuilder::stroke(px(1.5 * z));
                path.move_to(point(x(a), y0));
                path.line_to(point(x(b), y1));
                if let Ok(path) = path.build() {
                    window.paint_path(path, color(hue));
                }
            }
            let lane = row.lane.min(MAX_LANES - 1);
            let r = px(3.5 * z);
            window.paint_quad(
                fill(Bounds::new(point(x(lane) - r, mid - r), size(r * 2.0, r * 2.0)), color(lane))
                    .corner_radii(r),
            );
        },
    )
    .w(px(LANE_W * z * lanes.min(MAX_LANES) as f32))
    .h_full()
    .flex_shrink_0()
}
```

```rust
    fn restore_now(&mut self, at: GitAt, hash: String, then: String, now: String, short: String, window: &mut Window, cx: &mut Context<Self>) {
        let Some(repo) = self.git_at(&at).and_then(|g| g.repo().cloned()) else {
            return;
        };
        let trash = self.trash.clone();
        let done = format!("Restored {now} from {short}");
        let what = done.clone();
        self.act_job(
            at,
            move || -> Result<Option<Kept>, String> {
                let bytes = repo.file_at(&hash, &then).map_err(|e| e.to_string())?;
                let paths = [now.clone()];
                let (files, size) = trash::measure(&repo.root, &paths);
                let kept = match trash.as_ref().filter(|_| trash::fits(files, size)) {
                    Some(t) => Some(
                        t.put(&repo.root, &paths, &what, SystemTime::now())
                            .map_err(|e| format!("Couldn't keep a copy, so nothing was restored: {e}"))?,
                    ),
                    None => None,
                };
                let full = repo.root.join(&now);
                if let Some(dir) = full.parent() {
                    std::fs::create_dir_all(dir).map_err(|e| e.to_string())?;
                }
                std::fs::write(&full, bytes).map_err(|e| format!("Couldn't write {now}: {e}"))?;
                if let (Some(t), Some(k)) = (&trash, &kept) {
                    let _ = t.seal(k);
                    let _ = t.prune(SystemTime::now());
                }
                Ok(kept)
            },
            move |ws, at, result, window, cx| {
                let entity = cx.entity();
```

```rust
/// Split mode's rows: each hunk's line rows replaced by pairs. A context line is paired with
/// itself; a run of removed lines with the run of added lines after it, the i-th with the i-th;
/// what is left over sits beside an empty cell. A conflicted file keeps its rows (its parts
/// aren't an old and a new side); "No newline at end of file" notes go (`ponytail:` the
/// gutter can't say which side's: an end-of-file mark on the cell if people miss it).
pub(crate) fn split_rows(rows: Vec<Row>, files: &[Shown]) -> Vec<Row> {
    let mut out = Vec::with_capacity(rows.len());
    let mut paired: Option<(usize, usize)> = None;
    for row in rows {
        match row {
            Row::Line { file, hunk, .. } if paired == Some((file, hunk)) => {}
            Row::Note { file, note: Note::NoNewline } if paired.is_some_and(|(f, _)| f == file) => {}
            Row::Hunk { file, hunk } if files[file].diff.status != templum_git::Change::Conflict => {
                out.push(row);
                out.extend(pairs(file, hunk, &files[file].diff.hunks[hunk].lines));
                paired = Some((file, hunk));
            }
            other => {
                paired = None;
                out.push(other);
            }
        }
    }
    out
}

fn pairs(file: usize, hunk: usize, lines: &[templum_git::Line]) -> Vec<Row> {
    use templum_git::Kind;
    let mut out = Vec::new();
    let mut i = 0;
    while i < lines.len() {
        if lines[i].kind == Kind::Context {
            out.push(Row::Pair { file, hunk, old: Some(i), new: Some(i) });
            i += 1;
            continue;
        }
        let run = |from: usize, kind: Kind| (from..lines.len()).take_while(|&k| lines[k].kind == kind).collect::<Vec<_>>();
        let removed = run(i, Kind::Removed);
        let added = run(i + removed.len(), Kind::Added);
        for k in 0..removed.len().max(added.len()) {
            out.push(Row::Pair { file, hunk, old: removed.get(k).copied(), new: added.get(k).copied() });
        }
        i += removed.len() + added.len();
    }
    out
}
```

```rust
/// new number), an empty cell dimmed; each half `half` columns wide.
#[allow(clippy::too_many_arguments)]
fn pair_row(
    ws: &Workspace,
    id: BlockId,
    s: &Shown,
    file: usize,
    hunk: usize,
    old: Option<usize>,
    new: Option<usize>,
    ix: usize,
    row_h: f32,
    z: f32,
    cx: &mut Context<Workspace>,
) -> AnyElement {
    let half = ws.space.git(id).map_or(0, |g| g.diff().half);
    let cw = ws.cell.size.w * FONT / ws.cell.font_size * z;
    let half_w = px(cw * half as f32);
    let side = |line: Option<usize>, right: bool, cx: &mut Context<Workspace>| {
        line.map(|l| {
            let mut p = line_paint(ws, id, s, file, hunk, l, ix, right, row_h, z, cx);
            if right { p.old_no = None } else { p.new_no = None }
            p
        })
    };
    let (left, right) = (side(old, false, cx), side(new, true, cx));
    let empty = ws.ui.hover;
    let shaped = ws.space.git(id).map(|g| g.diff().shaped.clone());
    div()
        .id(("git-pair", ix))
        .w_full()
        .min_w(half_w * 2.0)
        .h(px(row_h))
        .child(
            canvas(
                |_, _, _| {},
                move |bounds, _, window, cx| {
                    for (paint, right, x0) in [(left.as_ref(), false, bounds.origin.x), (right.as_ref(), true, bounds.origin.x + half_w)] {
                        let cell = Bounds::new(point(x0, bounds.origin.y), size(half_w, bounds.size.height));
                        match paint {
                            Some(p) => {

                                if let Some(shaped) = &shaped {
                                    shaped.borrow_mut().insert((ix, right), Shaped { line, x, left: x0 });
                                }
                            }
                            None => window.paint_quad(fill(cell, empty)),
                        }
                    }
                },
            )
            .size_full(),
        )
        // The same mouse handlers as line_row: press, drag, right-click (Target::GitLine).
        .test_support()
        .into_any_element()
```
