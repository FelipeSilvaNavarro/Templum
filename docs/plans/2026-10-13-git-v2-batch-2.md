```rust
/// A `---`/`+++` line's path: unquoted, without its prefix and the tab git adds after a name
/// with spaces; None for /dev/null.
fn named(p: &str, prefix: &str) -> Option<String> {
    unquoted(p.trim_end_matches('\t'))
        .strip_prefix(prefix)
        .map(str::to_string)
}

/// A name as git prints it, C-quoted (`"a/we\"ird.txt"`) when it holds a quote, a backslash
/// or a control character, even with core.quotePath off: as written. A name not in quotes
/// comes back as it is.
fn unquoted(s: &str) -> String {
    let Some(inner) = s.strip_prefix('"').and_then(|s| s.strip_suffix('"')) else {
        return s.to_string();
    };
    let mut bytes = Vec::with_capacity(inner.len());
    let mut it = inner.bytes().peekable();
    while let Some(b) = it.next() {
        if b != b'\\' {
            bytes.push(b);
            continue;
        }
        match it.next() {
            Some(b'a') => bytes.push(7),
            Some(b'b') => bytes.push(8),
            Some(b't') => bytes.push(b'\t'),
            Some(b'n') => bytes.push(b'\n'),
            Some(b'v') => bytes.push(11),
            Some(b'f') => bytes.push(12),
            Some(b'r') => bytes.push(b'\r'),
            // \ooo: a byte in octal (a non-UTF-8 name, or quotePath on).
            Some(d @ b'0'..=b'7') => {
                let mut v = u32::from(d - b'0');
                for _ in 0..2 {
                    if let Some(&n @ b'0'..=b'7') = it.peek() {
                        v = v * 8 + u32::from(n - b'0');
                        it.next();
                    }
                }
                bytes.push(v as u8);
            }
            // \" and \\.
            Some(c) => bytes.push(c),
            None => {}
        }
    }
    String::from_utf8_lossy(&bytes).into_owned()
}

/// `"a/x" "b/x"`, `a/x "b/y"` or `"a/x" b/y`, unquoted; None when neither side is quoted.
fn quoted_halves(rest: &str) -> Option<(String, String)> {
    if rest.starts_with('"') {
        let end = closing_quote(rest)?;
        let (a, b) = rest.split_at(end + 1);
        return Some((unquoted(a), unquoted(b.trim_start())));
    }
    let i = rest.find(" \"b/")?;
    Some((rest[..i].to_string(), unquoted(&rest[i + 1..])))
}

/// The index of the quote that closes the one `s` starts with.
fn closing_quote(s: &str) -> Option<usize> {
    let bytes = s.as_bytes();
    let mut i = 1;
    while i < bytes.len() {
        match bytes[i] {
            b'\\' => i += 2,
            b'"' => return Some(i),
            _ => i += 1,
        }
    }
    None
}
```

After `parse_hunk_header`:

```rust
/// Hunk `hunk` of `file` alone, as a patch `git apply` takes: on the index forward (stage), on
/// the index reversed (unstage), on the working tree reversed (discard). It names only the
/// file's current path, so a staged rename stays staged when one of its hunks is unstaged.
/// None when the file can't be taken apart by hunk (`FileDiff::by_hunk`) or has no such hunk.
pub fn hunk_patch(file: &FileDiff, hunk: usize) -> Option<String> {
    if !file.by_hunk() {
        return None;
    }
```

```rust
/// The other way: a name as a patch must carry it for git to read it back.
fn quoted(s: &str) -> String {
    if !s.chars().any(|c| c == '"' || c == '\\' || c.is_control()) {
        return s.to_string();
    }
    let mut out = String::from('"');
    for c in s.chars() {
        match c {
            '"' => out.push_str("\\\""),
            '\\' => out.push_str("\\\\"),
            '\t' => out.push_str("\\t"),
            '\n' => out.push_str("\\n"),
            '\r' => out.push_str("\\r"),
            c if c.is_control() => {
                for b in c.to_string().bytes() {
                    out.push_str(&format!("\\{b:03o}"));
                }
            }
            c => out.push(c),
        }
    }
    out.push('"');
    out
}
```

```rust
    /// `git apply` of `patch` (a `hunk_patch`): on the index with `cached`, the other way with
    /// `reverse`. Written to a temporary file, since `run` gives git no stdin. A patch that no
    /// longer applies (the file changed since it was read) is git's error.
    pub fn apply_patch(&self, patch: &str, cached: bool, reverse: bool) -> Result<(), GitError> {
        static NEXT: AtomicU64 = AtomicU64::new(0);
        let file = std::env::temp_dir().join(format!(
            "templum-{}-{}.patch",
            std::process::id(),
            NEXT.fetch_add(1, Ordering::Relaxed)
        ));
        std::fs::write(&file, patch).map_err(|e| GitError {
            command: shown(&self.program, &["apply"]),
            code: None,
            stderr: format!("Couldn't write the patch: {e}"),
            missing: false,
        })?;
        let path = file.to_string_lossy().into_owned();
        // nowarn: a user's apply.whitespace=error would refuse a hunk with a trailing space.
        let mut args = vec!["apply", "--whitespace=nowarn"];
        if cached {
            args.push("--cached");
        }
        if reverse {
            args.push("--reverse");
        }
        args.push(&path);
        let result = self.run(&args).map(drop);
        let _ = std::fs::remove_file(&file);
        result
    }
```

**How it works:** one folder per discard under the trash's folder, named `{millis:013}-{pid}-{n}` so names sort by
time and never collide. `files/{i}` holds the i-th file as it was (`files/{i}.link` a symlink's target), `manifest`

one). `seal`, run after the discard, writes `after`: each file's state then (size and modification time, or `-`), so a
restore can tell a file changed since and ask. A restored entry leaves the trash.

```rust
use std::fs;
use std::io;
use std::path::{Path, PathBuf};
use std::sync::atomic::{AtomicU64, Ordering};
use std::time::{Duration, SystemTime, UNIX_EPOCH};

/// Past either, a discard keeps no copy, and the question says so before it is done.
pub const MAX_FILES: usize = 500;
pub const MAX_BYTES: u64 = 100 * 1024 * 1024;
/// Discards older than this, or past the newest KEEP, are deleted.
pub const MAX_AGE: Duration = Duration::from_secs(7 * 24 * 3600);
pub const KEEP: usize = 50;

/// The trash's folder (`trash/` in Templum's data folder).
#[derive(Clone, Debug)]
pub struct Trash {
    dir: PathBuf,
}

/// One discard, as kept.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Kept {
    /// Its folder's name: they sort by time.
    pub id: String,
    /// The repository's root.
    pub root: PathBuf,
    /// Unix seconds.
    pub time: u64,
    /// What was done: "Discarded src/x.rs".
    pub what: String,
    /// Each path, relative to the root, and whether it existed before the discard.
    pub files: Vec<(String, bool)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Restored {
    /// Put back; the discard left the trash.
    Done,
    /// These files changed since the discard, so nothing was written: `force` overwrites them.
    Changed(Vec<String>),
}

/// How many files a discard of `paths` keeps, and their size.
pub fn measure(root: &Path, paths: &[String]) -> (usize, u64) {
    let bytes = paths
        .iter()
        .filter_map(|p| fs::symlink_metadata(root.join(p)).ok())
        .map(|m| m.len())
        .sum();
    (paths.len(), bytes)
}

/// A discard this size gets a copy.
pub fn fits(files: usize, bytes: u64) -> bool {
    files <= MAX_FILES && bytes <= MAX_BYTES
}

/// A file's state, to tell later whether it changed: its size and modification time (a link
/// itself, not its target), or `-` when it doesn't exist.
fn stamp(path: &Path) -> String {
    match fs::symlink_metadata(path) {
        Ok(m) => {
            let mtime = m
                .modified()
                .ok()
                .and_then(|t| t.duration_since(UNIX_EPOCH).ok())
                .map_or(0, |d| d.as_nanos());
            format!("{}:{mtime}", m.len())
        }
        Err(_) => "-".to_string(),
    }
}

impl Trash {
    pub fn new(dir: PathBuf) -> Trash {
        Trash { dir }
    }

    /// Copies `paths` (relative to `root`) as they are now, before a discard. On a failure the
    /// half-made entry is removed: the caller must not discard then.
    pub fn put(&self, root: &Path, paths: &[String], what: &str, now: SystemTime) -> io::Result<Kept> {
        static NEXT: AtomicU64 = AtomicU64::new(0);
        let since = now.duration_since(UNIX_EPOCH).unwrap_or_default();
        let id = format!(
            "{:013}-{}-{}",
            since.as_millis(),
            std::process::id(),
            NEXT.fetch_add(1, Ordering::Relaxed)
        );
        let entry = self.dir.join(&id);
        let kept = Kept {
            id,
            root: root.to_path_buf(),
            time: since.as_secs(),
            what: what.to_string(),
            files: Vec::new(),
        };
        self.fill(&entry, kept, paths).inspect_err(|_| {
            let _ = fs::remove_dir_all(&entry);
        })
    }

    fn fill(&self, entry: &Path, mut kept: Kept, paths: &[String]) -> io::Result<Kept> {
        let files = entry.join("files");
        fs::create_dir_all(&files)?;
        for (i, path) in paths.iter().enumerate() {
            let from = kept.root.join(path);
            let meta = fs::symlink_metadata(&from).ok();
            match &meta {
                Some(m) if m.file_type().is_symlink() => {
                    let target = fs::read_link(&from)?;
                    fs::write(files.join(format!("{i}.link")), target.to_string_lossy().as_bytes())?;
                }
                Some(_) => {
                    fs::copy(&from, files.join(i.to_string()))?;
                }
                None => {}
            }
            kept.files.push((path.clone(), meta.is_some()));
        }
        let mut manifest = format!("{}\0{}\0{}\0", kept.root.to_string_lossy(), kept.time, kept.what);
        for (path, existed) in &kept.files {
            manifest.push_str(&format!("{path}\0{}\0", u8::from(*existed)));
        }
        fs::write(entry.join("manifest"), manifest)?;
        Ok(kept)
    }

    /// After the discard: each file's state then, so a restore can tell one changed since.
    pub fn seal(&self, kept: &Kept) -> io::Result<()> {
        let after: String = kept
            .files
            .iter()
            .map(|(path, _)| stamp(&kept.root.join(path)) + "\0")
            .collect();
        fs::write(self.dir.join(&kept.id).join("after"), after)
    }

    /// Every discard kept, newest first. One that can't be read is left out.
    pub fn list(&self) -> Vec<Kept> {
        let mut all: Vec<Kept> = fs::read_dir(&self.dir)
            .into_iter()
            .flatten()
            .flatten()
            .filter_map(|e| read_manifest(&e.path(), e.file_name().to_string_lossy().into_owned()))
            .collect();
        all.sort_by(|a, b| b.id.cmp(&a.id));
        all
    }

    /// Puts discard `id` back. Unless `force`, a file changed since the discard stops it,
    /// nothing written, and is named. Done, the discard leaves the trash.
    pub fn restore(&self, id: &str, force: bool) -> io::Result<Restored> {
        let entry = self.dir.join(id);
        let kept = read_manifest(&entry, id.to_string()).ok_or_else(|| {
            io::Error::new(io::ErrorKind::NotFound, "this discard is no longer in the trash")
        })?;
        if !force && let Ok(after) = fs::read_to_string(entry.join("after")) {
            let changed: Vec<String> = kept
                .files
                .iter()
                .zip(after.split('\0'))
                .filter(|((path, _), then)| stamp(&kept.root.join(path)) != *then)
                .map(|((path, _), _)| path.clone())
                .collect();
            if !changed.is_empty() {
                return Ok(Restored::Changed(changed));
            }
        }
        let files = entry.join("files");
        for (i, (path, existed)) in kept.files.iter().enumerate() {
            let to = kept.root.join(path);
            let link = files.join(format!("{i}.link"));
            let now = fs::symlink_metadata(&to).ok();
            let is_link = now.as_ref().is_some_and(|m| m.file_type().is_symlink());
            // A link is replaced, never written through; a file that didn't exist goes.
            if now.is_some() && (!existed || link.exists() || is_link) {
                fs::remove_file(&to)?;
            }
            if !existed {
                continue;
            }
            if let Some(dir) = to.parent() {
                fs::create_dir_all(dir)?;
            }
            if link.exists() {
                restore_link(&fs::read_to_string(&link)?, &to)?;
            } else {
                fs::copy(files.join(i.to_string()), &to)?;
            }
        }
        fs::remove_dir_all(&entry)?;
        Ok(Restored::Done)
    }

    pub fn remove(&self, id: &str) -> io::Result<()> {
        fs::remove_dir_all(self.dir.join(id))
    }

    /// Deletes the discards older than MAX_AGE and those past the newest KEEP; how many.
    pub fn prune(&self, now: SystemTime) -> io::Result<usize> {
        let now = now.duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
        let mut removed = 0;
        for (i, kept) in self.list().into_iter().enumerate() {
            if i >= KEEP || now.saturating_sub(kept.time) > MAX_AGE.as_secs() {
                self.remove(&kept.id)?;
                removed += 1;
            }
        }
        Ok(removed)
    }
}

fn read_manifest(entry: &Path, id: String) -> Option<Kept> {
    let text = fs::read_to_string(entry.join("manifest")).ok()?;
    let mut f = text.split('\0');
    let (root, time, what) = (f.next()?, f.next()?.parse().ok()?, f.next()?);
    let mut files = Vec::new();
    while let (Some(path), Some(existed)) = (f.next(), f.next()) {
        files.push((path.to_string(), existed == "1"));
    }
    Some(Kept {
        id,
        root: PathBuf::from(root),
        time,
        what: what.to_string(),
        files,
    })
}

#[cfg(unix)]
fn restore_link(target: &str, to: &Path) -> io::Result<()> {
    std::os::unix::fs::symlink(target, to)
}

/// Without symlinks (git's core.symlinks=false on Windows) a link is a file holding its target.
#[cfg(not(unix))]
fn restore_link(target: &str, to: &Path) -> io::Result<()> {
    fs::write(to, target)
}
```

```rust
    /// `git add --all`: every change, new and deleted files included (Commit all changes).
    pub fn stage_all(&self) -> Result<(), GitError> {
        self.run(&["add", "--all"]).map(drop)
    }

    /// The commit `short` is on the current branch's upstream: pushed, so it can't be taken
    /// back for others. False with no upstream, and on a detached HEAD.
    pub fn on_upstream(&self, short: &str) -> Result<bool, GitError> {
        match self.run(&["merge-base", "--is-ancestor", short, "@{u}"]) {
            Ok(_) => Ok(true),
            Err(e) if e.code == Some(1) => Ok(false),
            Err(e)
                if e.stderr.contains("no upstream")
                    || e.stderr.contains("does not point to a branch") =>
            {
                Ok(false)
            }
            Err(e) => Err(e),
        }
    }

    /// Takes the last commit back, its changes left staged, and gives its message (for the
    /// fields). `short` must still be HEAD: a commit made since is never the one undone. After an
    /// Amend (`amended`) the commit the amend replaced comes back (`HEAD@{1}`); the repository's
    /// first commit has no parent, so HEAD goes (`update-ref -d`).
    pub fn undo_commit(&self, short: &str, amended: bool) -> Result<String, GitError> {
        let now = self.text(&["rev-parse", "--verify", "-q", "HEAD"]).unwrap_or_default();
        if short.is_empty() || !now.trim().starts_with(short) {
            return Err(GitError {
                command: shown(&self.program, &["reset", "--soft", "HEAD~1"]),
                code: None,
                stderr: format!("error: the last commit is no longer {short}"),
                missing: false,
            });
        }
        let message = self.last_message()?;
        if amended {
            self.run(&["reset", "-q", "--soft", "HEAD@{1}"])?;
        } else if self.run(&["rev-parse", "--verify", "-q", "HEAD~1"]).is_ok() {
            self.run(&["reset", "-q", "--soft", "HEAD~1"])?;
        } else {
            self.run(&["update-ref", "-d", "HEAD"])?;
        }
        Ok(message)
    }
```

```rust
//! the notification, Recently discarded… to put one back later. See docs/specs/git.md,

use std::time::SystemTime;

use gpui_kit::component::notification::Notification;
use gpui_kit::{Context, Window};
use templum_canvas::BlockId;
use templum_git::GitError;

/// What a hunk's patch git refuses says: the file changed since it was read.
pub(crate) const REFUSED: &str = "The file changed; try again";

/// What a discard throws away: whole files' unstaged changes, or one hunk (its patch, reversed
/// on the working tree).
#[derive(Clone)]
pub(crate) enum Discarding {
    Files(Vec<templum_git::Entry>),
    Hunk(String),
}

/// Why a discard didn't happen.
enum Failed {
    /// The copy couldn't be kept, so nothing was discarded.
    Keep(std::io::Error),
    Git(GitError),
}

/// "1,240".
pub(crate) fn thousands(n: usize) -> String {
    let digits = n.to_string();
    let mut out = String::new();
    for (i, c) in digits.chars().enumerate() {
        if i > 0 && (digits.len() - i) % 3 == 0 {
            out.push(',');
        }
        out.push(c);
    }
    out
}

impl Workspace {
    /// Discard changes… and Discard all…: the unstaged changes of `paths`, after the question.
    pub(crate) fn discard(
        &mut self,
        id: BlockId,
        paths: Vec<String>,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let Some(git) = self.space.git(id) else {
            return;
        };
        let entries: Vec<templum_git::Entry> = git
            .status
            .unstaged()
            .into_iter()
            .filter(|e| paths.contains(&e.path))
            .cloned()
            .collect();
        let (title, done) = match entries.as_slice() {
            [] => return,
            [one] => (
                format!("Discard the changes to {}?", one.path),
                format!("Discarded {}", one.path),
            ),
            many => (
                format!("Discard {} files?", thousands(many.len())),
                format!("Discarded {} files", thousands(many.len())),
            ),
        };
        let paths = entries.iter().map(|e| e.path.clone()).collect();
        self.ask_discard(id, paths, title, done, Discarding::Files(entries), window, cx);
    }

    /// A hunk's Discard: `patch` reversed on `path`, after the question.
    pub(crate) fn discard_hunk(
        &mut self,
        id: BlockId,
        path: String,
        patch: String,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let title = format!("Discard this change to {path}?");
        let done = format!("Discarded a change to {path}");
        self.ask_discard(id, vec![path], title, done, Discarding::Hunk(patch), window, cx);
    }

    /// Measures what a copy would take (in the background: a build folder is thousands of
    /// files), then asks, saying whether the discard can be undone.
    #[allow(clippy::too_many_arguments)]
    fn ask_discard(
        &mut self,
        id: BlockId,
        paths: Vec<String>,
        title: String,
        done: String,
        what: Discarding,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let Some(root) = self
            .space
            .git(id)
            .and_then(|g| g.repo().map(|r| r.root.clone()))
        else {
            return;
        };
        let measured = paths.clone();
        self.git_job(
            self.here(id),
            move || trash::measure(&root, &measured),
            move |ws, at, (files, bytes), window, cx| {
                let keep = ws.trash.is_some() && trash::fits(files, bytes);
                let why = if keep {
                    let them = if paths.len() == 1 { "it" } else { "them" };
                    format!("You can restore {them} from Recently discarded.")
                } else if ws.trash.is_some() {
                    let size = if files > trash::MAX_FILES {
                        format!("{} files", thousands(files))
                    } else {
                        human(bytes)
                    };
                    format!("Too much to keep a copy: {size}. This can't be undone.")
                } else {
                    "This can't be undone.".to_string()
                };
                let entity = cx.entity();
                prompt::confirm(
                    title,
                    why,
                    "Discard",
                    move |window, cx| {
                        let (paths, done, what, at) =
                            (paths.clone(), done.clone(), what.clone(), at.clone());
                        entity.update(cx, |ws, cx| {
                            ws.discard_now(at, paths, done, what, keep, window, cx)
                        })
                    },
                    window,
                    cx,
                );
            },
            window,
            cx,
        );
    }

    /// The discard, in the background: the copy first (with `keep`), then git, then the copy's
    /// seal. A copy that can't be made stops it before anything is thrown away.
    #[allow(clippy::too_many_arguments)]
    fn discard_now(
        &mut self,
        at: GitAt,
        paths: Vec<String>,
        done: String,
        what: Discarding,
        keep: bool,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let Some(repo) = self.git_at(&at).and_then(|g| g.repo().cloned()) else {
            return;
        };
        let trash = self.trash.clone().filter(|_| keep);
        let hunk = matches!(what, Discarding::Hunk(_));
        let kept_as = done.clone();
        self.act_job(
            at,
            move || -> Result<Option<Kept>, Failed> {
                let kept = match &trash {
                    Some(t) => Some(
                        t.put(&repo.root, &paths, &kept_as, SystemTime::now())
                            .map_err(Failed::Keep)?,
                    ),
                    None => None,
                };
                let discarded = match &what {
                    Discarding::Files(entries) => repo.discard(&entries.iter().collect::<Vec<_>>()),
                    Discarding::Hunk(patch) => repo.apply_patch(patch, false, true),
                };
                if let Err(e) = discarded {
                    if let (Some(t), Some(k)) = (&trash, &kept) {
                        let _ = t.remove(&k.id);
                    }
                    return Err(Failed::Git(e));
                }
                if let (Some(t), Some(k)) = (&trash, &kept) {
                    let _ = t.seal(k);
                    let _ = t.prune(SystemTime::now());
                }
                Ok(kept)
            },
            move |ws, at, result, window, cx| {
                let entity = cx.entity();
                match result {
                    Ok(Some(kept)) => {
                        let (block, entry) = (at.block, kept.id);
                        let undo: Extra = (
                            "git-undo",
                            "Undo",
                            Rc::new(
                                move |ws: &mut Workspace,
                                      window: &mut Window,
                                      cx: &mut Context<Workspace>| {
                                    ws.restore_discarded(block, entry.clone(), false, window, cx)
                                },
                            ),
                        );
                        notify_with(
                            &entity,
                            at.clone(),
                            Notification::success(done),
                            vec![undo],
                            window,
                            cx,
                        );
                    }
                    Ok(None) => show_notification(window, cx, Notification::success(done)),
                    Err(Failed::Keep(e)) => {
                        let message = format!("Couldn't keep a copy, so nothing was discarded: {e}");
                        eprintln!("templum: {message}");
                        show_notification(window, cx, Notification::error(message));
                    }
                    Err(Failed::Git(e)) if hunk => notify_git_failure(
                        &entity,
                        at.clone(),
                        &e,
                        REFUSED.to_string(),
                        Vec::new(),
                        window,
                        cx,
                    ),
                    Err(Failed::Git(e)) => {
                        notify_git_error(&entity, at.clone(), &e, Vec::new(), window, cx)
                    }
                }
                ws.after_action(at, window, cx);
            },
            window,
            cx,
        );
    }

    /// Puts discard `entry` back (Undo, or Restore in Recently discarded…). A file changed since
    /// the discard asks before it is written over (`force`).
    pub(crate) fn restore_discarded(
        &mut self,
        id: BlockId,
        entry: String,
        force: bool,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let Some(trash) = self.trash.clone() else {
            return;
        };
        let job_entry = entry.clone();
        self.act_job(
            self.here(id),
            move || trash.restore(&job_entry, force),
            move |ws, at, result, window, cx| {
                match result {
                    Ok(Restored::Done) => {}
                    Ok(Restored::Changed(paths)) => {
                        let title = match paths.as_slice() {
                            [one] => format!("{one} changed since it was discarded"),
                            many => format!(
                                "{} files changed since they were discarded",
                                many.len()
                            ),
                        };
                        let (entity, block) = (cx.entity(), at.block);
                        prompt::confirm(
                            title,
                            "Restoring puts the discarded version back over the changes.",
                            "Restore",
                            move |window, cx| {
                                let entry = entry.clone();
                                entity.update(cx, |ws, cx| {
                                    ws.restore_discarded(block, entry, true, window, cx)
                                })
                            },
                            window,
                            cx,
                        );
                    }
                    Err(e) => {
                        let message = format!("Couldn't restore: {e}");
                        eprintln!("templum: {message}");
                        show_notification(window, cx, Notification::error(message));
                    }
                }
                ws.after_action(at, window, cx);
            },
            window,
            cx,
        );
    }

    /// Recently discarded…: this repository's discards still in the trash, read in the
    /// background, then listed.
    pub(crate) fn show_discarded(
        &mut self,
        id: BlockId,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let (Some(trash), Some(root)) = (
            self.trash.clone(),
            self.space
                .git(id)
                .and_then(|g| g.repo().map(|r| r.root.clone())),
        ) else {
            return;
        };
        self.git_job(
            self.here(id),
            move || {
                trash
                    .list()
                    .into_iter()
                    .filter(|k| k.root == root)
                    .collect::<Vec<Kept>>()
            },
            |ws, at, kept, window, cx| {
                let (entity, dim) = (cx.entity(), ws.ui.title_dim);
                git_block::discarded::open(entity, at.block, kept, dim, window, cx);
            },
            window,
            cx,
        );
    }
}
```

```rust
    /// Commit takes every change (`git add --all` first): nothing staged, something changed, no
    /// conflict (adding one would mark it resolved), no Amend.
    pub fn commit_all(&self) -> bool {
        !self.amend
            && self.status.staged().is_empty()
            && !self.status.unstaged().is_empty()
            && self.status.conflicts().is_empty()
    }

    /// "Commit", or "Commit all changes (5 files)".
    pub fn commit_label(&self) -> String {
        if !self.commit_all() {
            return "Commit".to_string();
        }
        match self.status.unstaged().len() {
            1 => "Commit all changes (1 file)".to_string(),
            n => format!("Commit all changes ({n} files)"),
        }
    }

    pub fn summary_len(&self, cx: &App) -> usize {
        self.summary.read(cx).value().chars().count()
    }

    pub fn summary_too_long(&self, cx: &App) -> bool {
        self.summary_len(cx) > SUMMARY_MAX
    }

    /// Commit is possible: something staged (or every change, see `commit_all`), a summary, an
    /// author, nothing running. With Amend, nothing staged is fine (the message alone changes).
    pub fn can_commit(&self, cx: &App) -> bool {
        (self.amend || !self.status.staged().is_empty() || self.commit_all())
            && !self.summary.read(cx).value().trim().is_empty()
            && self.author.is_some()
            && self.acting == 0
```

Replace `git_commit` with:

```rust
    /// Commits with the form's message: what is staged, or every change (`Git::commit_all`).
    /// "Committed abc1234" with Undo commit, and the fields clear; with `push`, then pushes (a
    /// branch with no upstream asks to publish, as Push does).
    pub(crate) fn git_commit(
        &mut self,
        id: BlockId,
        push: bool,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let Some(git) = self.space.git(id) else {
            return;
        };
        if !git.can_commit(cx) {
            return;
        }
        let (message, amend, all) = (git.message(cx), git.amend, git.commit_all());
        let Some(repo) = git.repo().cloned() else {
            return;
        };
        let at = self.here(id);
        self.act_job(
            at,
            move || {
                if all {
                    repo.stage_all()?;
                }
                repo.commit(&message, amend)
            },
            move |ws, at, result, window, cx| {
                let entity = cx.entity();
                // Only a commit that happened is pushed.
                let mut pushed = false;
                match result {
                    Ok(short) => {
                        if let Some(git) = ws.git_at(&at) {
                            git.amend = false;
                            let (summary, description) =
                                (git.summary.clone(), git.description.clone());
                            summary.update(cx, |s, cx| s.set_value("", window, cx));
```

```rust
    fn fill_message(&mut self, at: &GitAt, message: &str, window: &mut Window, cx: &mut Context<Self>) {

    /// Undo commit (the notification after a commit): takes it back, its changes staged and its
    /// message in the fields. Pushed meanwhile, it can't be taken back for others: Revert…
    /// instead, after saying why.
    pub(crate) fn undo_commit(
        &mut self,
        id: BlockId,
        short: String,
        amended: bool,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let Some(repo) = self.space.git(id).and_then(|g| g.repo().cloned()) else {
            return;
        };
        let job_short = short.clone();
        self.act_job(
            self.here(id),
            move || -> Result<Option<String>, GitError> {
                if repo.on_upstream(&job_short)? {
                    return Ok(None);
                }
                repo.undo_commit(&job_short, amended).map(Some)
            },
            move |ws, at, result, window, cx| {
                match result {
                    Ok(Some(message)) => {
                        if let Some(git) = ws.git_at(&at) {
                            git.amend = false;
                        }
                        ws.fill_message(&at, &message, window, cx);
                    }
                    Ok(None) => {
                        let (entity, block) = (cx.entity(), at.block);
                        prompt::confirm(
                            format!("Revert {short}?"),
                            "Pushed commits can't be removed for others; this adds a new commit that undoes it.",
                            "Revert",
                            move |window, cx| {
                                let short = short.clone();
                                entity.update(cx, |ws, cx| {
                                    ws.git_action(block, move |repo| repo.revert(&short), window, cx)
                                })
                            },
                            window,
                            cx,
                        );
                    }
                    Err(e) => {
                        let entity = cx.entity();
                        notify_git_error(&entity, at.clone(), &e, Vec::new(), window, cx)
                    }
                }
                ws.after_action(at, window, cx);
            },
            window,
            cx,
        );
    }
```

```rust
    /// The left column's rows: each list's header (Conflicts only when there are some), then its
    /// entries matching Filter files without case, by their index in the whole list (what the
    /// rows' ids, buttons and menus use). A filter left from before narrows nothing once the

    pub fn listed(&self, cx: &App) -> Vec<(List, Option<usize>)> {
        let query = if self.filtering() {
            self.files_filter.read(cx).value().trim().to_lowercase()
        } else {
            String::new()
        };
        let s = &self.status;
        let mut items = Vec::new();
        for (which, entries) in [
            (List::Conflicts, s.conflicts()),
            (List::Staged, s.staged()),
            (List::Unstaged, s.unstaged()),

            if which == List::Conflicts && entries.is_empty() {
                continue;
            }
            items.push((which, None));
            items.extend(
                entries
                    .iter()
                    .enumerate()
                    .filter(|(_, e)| query.is_empty() || e.path.to_lowercase().contains(&query))
                    .map(|(i, _)| (which, Some(i))),
            );
        }
        items
    }
```

**Focus, read from GPUI 0.3.8:** on a mouse down, the innermost element tracking a focus handle takes the focus
(`elements/div.rs:2774`), then the block's root runs `focus_block`, which calls `Body::focus_inner`. So a click in

the block's handle otherwise. Keys go from the focused element up to the workspace's root: the body's `on_key_down`
sees them first, takes the spec's keys only while the block's own handle (not a field) is focused, and stops them;
everything else goes on to the workspace's shortcuts.

**The cursor:** a hunk, kept as (section, path, index) so it survives a read: after `s` stages hunk 1 of three, the
file's hunk 1 is the former hunk 2, the next one to review. When the index is gone, the file's last hunk; when the
file is gone, none. `↑`/`↓` clear it (then `s`, `u`, `d` act on the selected file); `[`/`]` set it and select its

```rust
//! A Git block's keys while it has the keyboard, its own focus and not a field's
//! (docs/specs/git.md, section 1, Keyboard).

use gpui_kit::{Context, KeyDownEvent, ScrollStrategy, Window};
use templum_canvas::BlockId;

use crate::git_block::List;
use crate::git_block::diff_view::Row;

use super::Workspace;

/// The hunk under the keyboard's cursor: its section, its path, its patch (None when it can't
/// go alone: a new or deleted file, whitespace ignored), its first line.
struct AtCursor {
    staged: bool,
    path: String,
    patch: Option<String>,
    line: u32,
}

impl Workspace {
    pub(crate) fn git_key(
        &mut self,
        id: BlockId,
        e: &KeyDownEvent,
        window: &mut Window,
        cx: &mut Context<Self>,
    ) {
        let Some(git) = self.space.git(id) else {
            return;
        };
        // A field has the keyboard: its letters are typing.
        if !git.focus.is_focused(window) {
            return;
        }
        let summary = git.summary.clone();
        let k = &e.keystroke;
        let m = &k.modifiers;
        if k.key == "enter" && m.secondary() && !m.shift && !m.alt {
            self.git_commit(id, false, window, cx);
        } else if m.control || m.platform || m.function {
            // The workspace's shortcuts.
            return;
        } else {
            match k.key.as_str() {
                "up" => self.step_file(id, -1, cx),
                "down" => self.step_file(id, 1, cx),
                "[" => self.step_hunk(id, -1, cx),
                "]" => self.step_hunk(id, 1, cx),
                "s" => self.key_stage(id, false, window, cx),
                "u" => self.key_stage(id, true, window, cx),
                "d" => self.key_discard(id, window, cx),
                "enter" => self.key_open(id, window, cx),
                "c" => summary.update(cx, |s, cx| s.focus(window, cx)),
                _ => return,
            }
        }
        cx.stop_propagation();
    }

    /// ↑ / ↓: the previous or next file of the left column. The panel scrolls to it, and the
    /// hunk cursor goes: `s`, `u` and `d` then act on the file.
    fn step_file(&mut self, id: BlockId, step: isize, cx: &mut Context<Self>) {
        let Some(git) = self.space.git(id) else {
            return;
        };
        let (conflicts, staged, unstaged) = (
            git.status.conflicts(),
            git.status.staged(),
            git.status.unstaged(),
        );
        let files: Vec<(usize, List, String)> = git
            .listed(cx)
            .into_iter()
            .enumerate()
            .filter_map(|(ix, (list, i))| {
                let entries = match list {
                    List::Conflicts => &conflicts,
                    List::Staged => &staged,
                    List::Unstaged => &unstaged,
                };
                Some((ix, list, entries.get(i?)?.path.clone()))
            })
            .collect();
        if files.is_empty() {
            return;
        }
        let at = git
            .selected
            .as_ref()
            .and_then(|(l, p)| files.iter().position(|(_, list, path)| list == l && path == p));
        let next = match at {
            Some(at) => at.saturating_add_signed(step).min(files.len() - 1),
            None if step > 0 => 0,
            None => files.len() - 1,
        };
        let (row, list, path) = files[next].clone();
        let Some(git) = self.space.git_mut(id) else {
            return;
        };
        git.panel.cursor = None;
        git.lists_scroll.scroll_to_item(row, ScrollStrategy::Center);
        let staged = list == List::Staged;
        if let Some(ix) = git.panel.row_of(staged, &path) {
            git.panel.scroll.scroll_to_item(ix, ScrollStrategy::Top);
        }
        git.selected = Some((list, path));
        cx.notify();
    }

    /// [ / ]: the keyboard's hunk moves to the previous or next hunk of the panel (from the
    /// selected file when there is none yet), selects its file, and the panel scrolls to it.
    fn step_hunk(&mut self, id: BlockId, step: isize, cx: &mut Context<Self>) {
        let Some(git) = self.space.git_mut(id) else {
            return;
        };
        let target = {
            let p = &git.panel;
            let hunks: Vec<usize> = p
                .rows
                .iter()
                .enumerate()
                .filter(|(_, r)| matches!(r, Row::Hunk { .. }))
                .map(|(ix, _)| ix)
                .collect();
            if hunks.is_empty() {
                return;
            }
            let next = match p.cursor_row().and_then(|row| hunks.iter().position(|&h| h == row)) {
                Some(at) => at.saturating_add_signed(step).min(hunks.len() - 1),
                None => {
                    let from = git
                        .selected
                        .as_ref()
                        .and_then(|(list, path)| p.row_of(*list == List::Staged, path))
                        .unwrap_or(0);
                    if step > 0 {
                        hunks.iter().position(|&h| h > from).unwrap_or(0)
                    } else {
                        hunks.iter().rposition(|&h| h < from).unwrap_or(hunks.len() - 1)
                    }
                }
            };
            let row = hunks[next];
            let Row::Hunk { file, hunk } = p.rows[row] else {
                return;
            };
            let s = &p.files[file];
            (row, s.staged, s.diff.path().to_string(), hunk)
        };
        let (row, staged, path, hunk) = target;
        git.panel.cursor = Some((staged, path.clone(), hunk));
        git.selected = Some((if staged { List::Staged } else { List::Unstaged }, path));
        git.panel.scroll.scroll_to_item(row, ScrollStrategy::Center);
        cx.notify();
    }

    fn at_cursor(&self, id: BlockId) -> Option<AtCursor> {
        let git = self.space.git(id)?;
        let p = &git.panel;
        let Row::Hunk { file, hunk } = p.rows[p.cursor_row()?] else {
            return None;
        };
        let s = &p.files[file];
        Some(AtCursor {
            staged: s.staged,
            path: s.diff.path().to_string(),
            patch: (!p.ignore_ws)
                .then(|| templum_git::hunk_patch(&s.diff, hunk))
                .flatten(),
            line: s.diff.hunks[hunk].new_start.max(1),
        })
    }

    /// s / u: the cursor's hunk (Stage in Changes, Unstage in Staged; one that can't go alone
    /// takes its whole file), else the selected file.
    fn key_stage(&mut self, id: BlockId, unstage: bool, window: &mut Window, cx: &mut Context<Self>) {
        let selected = self.space.git(id).and_then(|g| g.selected.clone());
        match self.at_cursor(id) {
            // s on a staged hunk, u on an unstaged one: nothing to do.
            Some(c) if c.staged != unstage => {}
            Some(AtCursor { patch: Some(patch), .. }) => {
                self.apply_hunk(id, patch, unstage, window, cx)
            }
            Some(c) if unstage => self.unstage(id, vec![c.path], window, cx),
            Some(c) => self.stage(id, vec![c.path], window, cx),
            None => match (selected, unstage) {
                (Some((List::Staged, path)), true) => self.unstage(id, vec![path], window, cx),
                (Some((List::Unstaged | List::Conflicts, path)), false) => {
                    self.stage(id, vec![path], window, cx)
                }
                _ => {}
            },
        }
    }

    /// d: the cursor's hunk in Changes (one that can't go alone takes its file), else the
    /// selected file's unstaged changes. Both ask first.
    fn key_discard(&mut self, id: BlockId, window: &mut Window, cx: &mut Context<Self>) {
        let selected = self.space.git(id).and_then(|g| g.selected.clone());
        match self.at_cursor(id) {
            Some(c) if c.staged => {}
            Some(AtCursor { path, patch: Some(patch), .. }) => {
                self.discard_hunk(id, path, patch, window, cx)
            }
            Some(c) => self.discard(id, vec![c.path], window, cx),
            None => {
                if let Some((List::Unstaged, path)) = selected {
                    self.discard(id, vec![path], window, cx)
                }
            }
        }
    }

    /// Enter: the file in an editor beside the block, at the cursor's hunk, else at the selected
    /// file's first.
    fn key_open(&mut self, id: BlockId, window: &mut Window, cx: &mut Context<Self>) {
        let selected = self.space.git(id).and_then(|g| g.selected.clone());
        match (self.at_cursor(id), selected) {
            (Some(c), _) => self.open_from_git_at(id, &c.path, Some(c.line), window, cx),
            (None, Some((_, path))) => self.open_from_git(id, &path, window, cx),
            (None, None) => {}
        }
    }
}
```
